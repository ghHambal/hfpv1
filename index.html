🚀 Hifz Tracker Pro

Master Deployment Guide (คู่มือทำสำเนาระบบสำหรับโรงเรียนใหม่)

1

สร้าง Project ใหม่ใน Supabase

  • ไปที่ Supabase.com และคลิก New Project
  • ตั้งชื่อ Project (เช่น hifz-school-name) และตั้งรหัสผ่าน Database Password ให้จำได้
  • เลือก Region เป็น Singapore (ap-southeast-1) (เพื่อความรวดเร็วในไทย)
  • รอประมาณ 2-3 นาทีจนกว่า Project จะสร้างเสร็จ
2

สร้างตารางฐานข้อมูล (SQL Script)

ไปที่เมนู SQL Editor (ไอคอน >_ แถบซ้าย) ➔ กด New Query ➔ วางโค้ดด้านล่าง แล้วกด RUN

-- 1. สร้างตารางการตั้งค่า (Settings)
CREATE TABLE IF NOT EXISTS "Settings" (
  "key" TEXT PRIMARY KEY,
  "value" TEXT NOT NULL
);

INSERT INTO "Settings" ("key", "value") VALUES 
('schoolName', 'ระบุชื่อโรงเรียน'),
('appName', 'Hifz Tracker Pro'),
('currentTerm', '1/2569')
ON CONFLICT ("key") DO NOTHING;

-- 2. สร้างตารางโปรแกรม (Programs)
CREATE TABLE IF NOT EXISTS "Programs" (
  "programId" TEXT PRIMARY KEY,
  "programName" TEXT NOT NULL,
  "targetJuzPerTerm" NUMERIC DEFAULT 0,
  "hasAssessment" BOOLEAN DEFAULT true,
  "isActive" BOOLEAN DEFAULT true
);

-- 3. สร้างตารางครู/บุคลากร (Teachers)
CREATE TABLE IF NOT EXISTS "Teachers" (
  "teacherId" TEXT PRIMARY KEY,
  "firstName" TEXT NOT NULL,
  "lastName" TEXT NOT NULL,
  "classRoom" TEXT,
  "phone" TEXT,
  "email" TEXT,
  "role" TEXT NOT NULL,
  "isActive" BOOLEAN DEFAULT true,
  "photoUrl" TEXT
);

-- สร้างบัญชีแอดมินเริ่มต้น

-- 4. สร้างตารางผู้ปกครอง (Parents)
CREATE TABLE IF NOT EXISTS "Parents" (
  "parentId" TEXT PRIMARY KEY,
  "firstName" TEXT NOT NULL,
  "lastName" TEXT NOT NULL,
  "phone" TEXT,
  "email" TEXT,
  "studentIds" TEXT,
  "isActive" BOOLEAN DEFAULT true,
  "photoUrl" TEXT
);

-- 5. สร้างตารางนักเรียน (Students)
CREATE TABLE IF NOT EXISTS "Students" (
  "studentId" TEXT PRIMARY KEY,
  "firstName" TEXT NOT NULL,
  "lastName" TEXT NOT NULL,
  "classRoom" TEXT,
  "teacherId" TEXT REFERENCES "Teachers"("teacherId") ON DELETE SET NULL,
  "programId" TEXT REFERENCES "Programs"("programId") ON DELETE SET NULL,
  "parentId" TEXT REFERENCES "Parents"("parentId") ON DELETE SET NULL,
  "pin4" TEXT,
  "isActive" BOOLEAN DEFAULT true,
  "photoUrl" TEXT
);

-- 6. สร้างตารางบันทึกการท่อง (Logs)
CREATE TABLE IF NOT EXISTS "Logs" (
  "logId" TEXT PRIMARY KEY,
  "studentId" TEXT NOT NULL REFERENCES "Students"("studentId") ON DELETE CASCADE,
  "teacherId" TEXT REFERENCES "Teachers"("teacherId") ON DELETE SET NULL,
  "date" DATE NOT NULL,
  "logType" TEXT NOT NULL,
  "juzNumber" INTEGER DEFAULT 0,
  "surahNumber" INTEGER DEFAULT 0,
  "surahName" TEXT,
  "ayah" TEXT,
  "score" NUMERIC DEFAULT 0,
  "pages" NUMERIC DEFAULT 0,
  "status" TEXT NOT NULL,
  "note" TEXT,
  "created_at" TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 7. สร้างตารางข้อมูลซูเราะห์ (Mapping & Surahs)
CREATE TABLE IF NOT EXISTS "Mapping" (
  "id" SERIAL PRIMARY KEY,
  "surahNumber" INTEGER NOT NULL,
  "surahNameThai" TEXT,
  "surahNameArabic" TEXT,
  "ayahRange" TEXT,
  "ayahStart" INTEGER,
  "ayahEnd" INTEGER,
  "pageRange" TEXT,
  "pageStart" INTEGER,
  "pageEnd" INTEGER,
  "juzNumber" INTEGER
);

CREATE TABLE IF NOT EXISTS "Surahs" (
  "number" INTEGER PRIMARY KEY,
  "name" TEXT,
  "englishName" TEXT,
  "numberOfAyahs" INTEGER,
  "juz" INTEGER
);

3

ตั้งค่าพื้นที่เก็บรูปภาพ (Storage Bucket)

หากโรงเรียนต้องการเก็บรูปโปรไฟล์ไว้ใน Supabase ให้ทำตามขั้นตอนนี้ (ถ้าใช้ Google Drive ข้ามข้อนี้ได้เลย)

  1. ไปที่เมนู Storage (ไอคอนรูปกล่อง แถบซ้าย)
  2. คลิก New Bucket
  3. ตั้งชื่อ Bucket ว่า avatars
  4. ✅ ติ๊กเลือก Public bucket แล้วกด Save
  5. ไปที่แท็บ Policies ในหน้า Storage ➔ ค้นหาชื่อ bucket "avatars"
  6. คลิก New Policy ➔ เลือก For full customization
  7. ตั้งชื่อ Policy ว่า Allow Public Access
  8. ตรงช่อง Allowed Operations ให้ติ๊ก SELECT, INSERT, UPDATE, DELETE
  9. กด Save policy เพื่ออนุญาตให้ระบบใช้อัปโหลดรูปได้
4

ดึง API Key ไปใส่ในเว็บแอป (App Script)

  • ไปที่เมนู Project Settings (ไอคอนฟันเฟืองล่างซ้าย) ➔ เลือก API
  • คัดลอก Project URL และ Project API keys (anon / public)
  • เปิดไฟล์ index.html (ใน Google Apps Script ของโรงเรียนใหม่) แล้วนำไปแทนที่ด้านบนสุดของโค้ด:
// ============================================
// 🚀 ตั้งค่าการเชื่อมต่อ Supabase (สำหรับโรงเรียนใหม่)
// ============================================
const SUPABASE_URL = 'วาง_URL_ที่นี่';
const SUPABASE_KEY = 'วาง_API_KEY_ที่นี่';
const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

🎉 เสร็จสิ้น! ระบบพร้อมใช้งานทันที
